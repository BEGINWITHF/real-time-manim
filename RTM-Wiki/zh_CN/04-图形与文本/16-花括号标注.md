# 花括号标注

"在对象旁边画一个花括号 + 注释文字"的标注族，定义在 `mobject/svg/brace.py`；手册条目为 `BraceText`（无 `BraceWithText`）。每个类各一条样例，样例里把该类**全部可填参数**各改一次。

> **`get_text` / `get_tex` 的 kwargs 只传给 `put_at_tip`**：`Brace.get_text(*text, **kwargs)` 里 `**kwargs` 全部喂给 `put_at_tip(mob, **kwargs)`（最终是 `mob.next_to(..., **kwargs)`），所以 `get_text("x", font_size=20)` / `get_tex("x", font_size=20)` 会抛 `TypeError: Mobject.next_to got an unexpected keyword argument 'font_size'`。字号/颜色要**取回对象后**再设置：`lbl = br.get_text("x"); lbl.scale(0.6).set_color(GREY)`；只有 `buff`、`aligned_edge` 这类 `next_to` 的参数能直接传。

## 参数

### `Brace`

```python
def __init__(self, mobject: Mobject, direction: Vector3DLike = DOWN, buff: float = 0.2, sharpness: float = 2, stroke_width: float = 0, fill_opacity: float = 1.0, background_stroke_width: float = 0, background_stroke_color: ParsableManimColor = BLACK, **kwargs) -> None:
```

| 参数 | 类型 | 默认值 | 说明 |
| --- | --- | --- | --- |
| `mobject` | `Mobject` | 必填 | 被标注的对象（花括号自动取它的位置与长度） |
| `direction` | `array-like` | `DOWN` | 花括号朝向；样例把 `DOWN`/`UP`/`LEFT`/`RIGHT` 四个方向都画一遍 |
| `buff` | `float` | `0.2` | 与对象的间距；样例 `0.9` 与默认对比 |
| `sharpness` | `float` | `2` | 拐角尖锐度；样例 `6`（尖端更尖、过渡更陡） |
| `stroke_width` | `float` | `0` | 描边宽度；样例 `2` |
| `fill_opacity` | `float` | `1.0` | 填充不透明度；样例 `0.5` |
| `background_stroke_width` | `float` | `0` | 背景描边宽度；样例 `10`（外圈更粗） |
| `background_stroke_color` | `ManimColor` | `BLACK` | 背景描边色；样例 `RED` |

方法：`get_text(*text, **kwargs)`（尖端放一个 `Tex` 并返回它）、`get_tex(*tex, **kwargs)`（同上但用 `MathTex`）、`put_at_tip(mob, use_next_to=True, **kwargs)`、`get_tip()`（尖端坐标）、`get_direction()`（朝向向量）。

#### 样例：四个朝向 / `buff`+`sharpness` / 描边填充 / 五个方法

```python
import os
import subprocess
import sys
from real_time_manim import (
    Scene, Brace, BraceText, BraceLabel, BraceBetweenPoints, ArcBrace,
    Square, Line, Dot, Text, Arc, VGroup, MathTex, DEGREES,
    BLUE, GREEN, RED, ORANGE, YELLOW, PURPLE, TEAL, GOLD, GREY_B, WHITE,
    UP, DOWN, LEFT, RIGHT, ORIGIN,
    Create, FadeIn, FadeOut, GrowFromCenter, Write,
)
from real_time_manim.vulkan_bind import MLWindow, Wait
from real_time_manim.record import fast_record_scene

class BraceDemo(Scene):
    def construct(self):
        win = MLWindow(960, 540)   #创建一个 RTM 窗口
        win.scene = self   #将名为 “self” 的 manim 窗口，绑定给名为 “win” 的 RTM 窗口
        title = Text("16-Brace", font_size=30, color=YELLOW).to_edge(UP)   #页面标题
        win.play(Write(title), run_time=0.5)   #写出标题

        # —— direction：花括号在对象的哪一侧 ——
        group = VGroup()
        for i, (d, name) in enumerate([(DOWN, "DOWN"), (UP, "UP"), (LEFT, "LEFT"), (RIGHT, "RIGHT")]):   #参数 “direction”
            sq = Square(side_length=1.2, color=BLUE).move_to([-4.8 + i * 3.2, 1.1, 0])
            br = Brace(sq, direction=d, color=GREEN)   #参数 “mobject / direction”
            lbl = br.get_text(name, buff=0.06)   #方法 “get_text”：kwargs 只会传给 put_at_tip（即 next_to 的 buff）
            lbl.scale(0.55).set_color(GREY_B)   #Tex 的字号/颜色要事后设（get_text 不吃这些 kwargs）
            group.add(sq, br, lbl)
            win.play(Create(sq), GrowFromCenter(br), FadeIn(lbl), run_time=0.45)   #四种朝向依次出现
        win.play(FadeOut(group), run_time=0.3)

        # —— buff / sharpness：与对象的间距、拐角尖锐度 ——
        sq2 = Square(side_length=1.4, color=BLUE).move_to([-2.0, 0.6, 0])
        br2 = Brace(sq2, direction=DOWN, buff=0.9, sharpness=6, color=ORANGE)   #参数 “buff / sharpness”
        lbl2 = br2.get_text("buff=0.9  sharpness=6")   #方法 “get_text”
        lbl2.scale(0.6).set_color(GREY_B)
        win.play(Create(sq2), GrowFromCenter(br2), FadeIn(lbl2), run_time=0.6)
        win.play(FadeOut(sq2), FadeOut(br2), FadeOut(lbl2), run_time=0.25)

        # —— stroke_width / fill_opacity / background_stroke_width / background_stroke_color ——
        ln = Line(LEFT * 2.2, RIGHT * 2.2, color=BLUE)
        br3 = Brace(ln, direction=DOWN, stroke_width=2, fill_opacity=0.5,
                    background_stroke_width=10, background_stroke_color=RED)   #参数 “stroke_width / fill_opacity / background_*”
        lbl3 = br3.get_text("stroke 2 - fill 0.5 - bg 10 - RED")   #方法 “get_text”（Tex 是数学模式：字符串里别带下划线）
        lbl3.scale(0.5).set_color(GREY_B)
        win.play(Create(ln), GrowFromCenter(br3), FadeIn(lbl3), run_time=0.7)
        win.play(FadeOut(ln), FadeOut(lbl3), run_time=0.25)

        # —— 方法：get_tex / put_at_tip / get_tip / get_direction ——
        lbl4 = br3.get_tex(r"x^2+y^2=r^2")   #方法 “get_tex”：改用 MathTex 标签
        lbl4.scale(0.7).set_color(TEAL)
        win.play(FadeIn(lbl4), run_time=0.4)
        tip = br3.get_tip()   #方法 “get_tip()”：拿到尖端坐标
        dot = Dot(tip, color=YELLOW, radius=0.07)
        dirn = br3.get_direction()   #方法 “get_direction()”：拿到朝向向量
        cap = Text("tip + direction = (%d, %d)" % (round(dirn[0]), round(dirn[1])),
                   font_size=20, color=GREY_B).next_to(lbl4, DOWN, buff=0.25)
        win.play(FadeIn(dot), FadeIn(cap), run_time=0.4)
        win.play(FadeOut(lbl4), run_time=0.25)   #尖端标签和 get_tex 标签同一位置，先收掉旧的
        tag = Text("put_at_tip", font_size=22, color=PURPLE)
        br3.put_at_tip(tag)   #方法 “put_at_tip(mob)”：把对象放到尖端
        win.play(FadeIn(tag), run_time=0.4)
        win.play(Wait(0.4))   #停留看结果
        win.close()   #关闭 RTM 窗口


# ===== 样例 2：BraceText —— 花括号 + 普通文本一步到位 =====================
```

这段代码在做什么：第一段把同一个正方形换四个 `direction` 各画一个花括号（每个花括号用 `get_text` 写上方向名）；第二段把 `buff` 拉到 0.9、`sharpness` 拉到 6，花括号明显离得更远、尖端更锐；第三段给一条横线加花括号并同时改 `stroke_width`（2）、`fill_opacity`（0.5）、`background_stroke_width`（10）、`background_stroke_color`（RED）；最后一段演示方法——`get_tex` 换成公式标签、`get_tip()` 拿尖端坐标画个黄点、`get_direction()` 把朝向向量打成读数、`put_at_tip()` 把另一个对象摆到尖端。

<video src="assets/RTM output/16-花括号标注-brace.mp4" controls width="560"></video>

> 提醒：`sharpness` 太小（如 1）时尖端会变钝、看起来像圆弧；`background_stroke_*` 画的是花括号**外圈**的一层粗边（`stroke_width` 是内圈），所以两层颜色可以不同。

### `BraceText`

```python
def __init__(self, obj: Mobject, text: str, label_constructor: type[SingleStringMathTex | Text] = Text, **kwargs) -> None:
```

| 参数 | 类型 | 默认值 | 说明 |
| --- | --- | --- | --- |
| `obj` | `Mobject` | 必填 | 被标注对象 |
| `text` | `str \| Mobject` | 必填 | 标注文字 |
| `label_constructor` | `类` | `Text` | 文字渲染类；样例里换成 `MathTex` 对比（`MathTex` 走 LaTeX，公式更端正） |
| `brace_direction` | `array-like` | `DOWN` | 花括号朝向；样例 `RIGHT` / `UP` |
| `font_size` | `float` | `DEFAULT_FONT_SIZE` | 标签字号；样例 `24` / `60` |
| `buff` | `float` | `0.2` | 花括号与对象的间距；样例 `0.7` |
| `brace_config` | `dict \| None` | `None` | 传给内部 `Brace` 的样式；样例 `{"color": ORANGE}` |

是 `BraceLabel` 的 `Text` 版：一次性构造"花括号 + 普通文本"，文字自动放在尖端外侧。

### `BraceLabel`

```python
def __init__(self, obj: Mobject, text: str, brace_direction: Vector3DLike = DOWN, label_constructor: type[SingleStringMathTex | Text] = MathTex, font_size: float = DEFAULT_FONT_SIZE, buff: float = 0.2, brace_config: dict | None = None, **kwargs) -> None:
```

| 参数 | 类型 | 默认值 | 说明 |
| --- | --- | --- | --- |
| `obj` | `Mobject` | 必填 | 被标注对象 |
| `text` | `str \| Mobject` | 必填 | 标签内容（字符串走 LaTeX，所以默认能直接写公式） |
| `brace_direction` | `array-like` | `DOWN` | 花括号朝向；样例 `LEFT` |
| `label_constructor` | `类` | `MathTex` | 标签渲染类（`BraceText` 就是把它换成 `Text`） |
| `font_size` | `float` | `DEFAULT_FONT_SIZE` | 标签字号；样例 `30` |
| `buff` | `float` | `0.2` | 间距 |
| `brace_config` | `dict \| None` | `None` | 花括号样式；样例 `{"color": TEAL}` |

方法：`creation_anim(label_anim=FadeIn, brace_anim=GrowFromCenter)`（花括号与标签一起出场的动画组）、`change_label(*text)`（只换标签）、`change_brace_label(obj, *text)`（换对象 + 换标签）、`shift_brace(obj)`（按对象重新贴一次花括号）。

#### 样例：默认 / `creation_anim` / `change_label` / `shift_brace` / `change_brace_label`

```python
import os
import subprocess
import sys
from real_time_manim import (
    Scene, Brace, BraceText, BraceLabel, BraceBetweenPoints, ArcBrace,
    Square, Line, Dot, Text, Arc, VGroup, MathTex, DEGREES,
    BLUE, GREEN, RED, ORANGE, YELLOW, PURPLE, TEAL, GOLD, GREY_B, WHITE,
    UP, DOWN, LEFT, RIGHT, ORIGIN,
    Create, FadeIn, FadeOut, GrowFromCenter, Write,
)
from real_time_manim.vulkan_bind import MLWindow, Wait
from real_time_manim.record import fast_record_scene

class BraceLabelDemo(Scene):
    def construct(self):
        win = MLWindow(960, 540)   #创建一个 RTM 窗口
        win.scene = self   #将名为 “self” 的 manim 窗口，绑定给名为 “win” 的 RTM 窗口
        title = Text("16-BraceLabel", font_size=30, color=YELLOW).to_edge(UP)   #页面标题
        win.play(Write(title), run_time=0.5)   #写出标题

        # —— 默认：label_constructor=MathTex ——
        sq = Square(side_length=2.0, color=BLUE).move_to([0, 0.6, 0])
        bl = BraceLabel(sq, r"2 \times 2")   #参数 “obj / text”
        win.play(Create(sq), bl.creation_anim(), run_time=0.6)   #方法 “creation_anim”

        # —— change_label：换掉标签文字 ——
        old_label = bl.label   #入场动画把旧标签单独加进了场景：换标签时要一起收掉

        win.play(FadeOut(old_label), FadeIn(bl.label), run_time=0.5)

        # —— shift_brace：花括号换到另一侧（同时文字跟着走）——
        bl2 = BraceLabel(sq, "LEFT", brace_direction=LEFT, font_size=30,
                         brace_config={"color": TEAL})   #参数 “brace_direction / font_size / brace_config”
        bl.shift_brace(sq)   #方法 “shift_brace(obj)”：按对象重新贴一次
        win.play(GrowFromCenter(bl2), run_time=0.6)   #左侧再来一个做对比

        # —— change_brace_label：换对象 + 换标签 ——
        sq2 = Square(side_length=1.4, color=GREEN).move_to([-4.4, -2.0, 0])
        bl2.change_brace_label(sq2, "moved")   #方法 “change_brace_label(obj, *text)”
        win.play(Create(sq2), run_time=0.5)
        win.play(Wait(0.5))   #停留看结果
        win.close()   #关闭 RTM 窗口
```

### `BraceBetweenPoints`

```python
def __init__(self, point_1: Point3DLike, point_2: Point3DLike, direction: Vector3DLike = ORIGIN, **kwargs) -> None:
```

| 参数 | 类型 | 默认值 | 说明 |
| --- | --- | --- | --- |
| `point_1` | `array-like` | 必填 | 端点 1 |

#### 样例：横向 `DOWN`/`UP` + 公式标签 + 竖直 `LEFT`

```python

import subprocess
import sys
from real_time_manim import (
    Scene, Brace, BraceText, BraceLabel, BraceBetweenPoints, ArcBrace,
    Square, Line, Dot, Text, Arc, VGroup, MathTex, DEGREES,

    UP, DOWN, LEFT, RIGHT, ORIGIN,
    Create, FadeIn, FadeOut, GrowFromCenter, Write,

from real_time_manim.vulkan_bind import MLWindow, Wait
from real_time_manim.record import fast_record_scene

        lbl3 = bb3.get_text("dy")
        lbl3.scale(0.6).set_color(GREY_B)
        win.play(FadeIn(Dot(c, color=YELLOW)), FadeIn(Dot(d, color=YELLOW)),
                 GrowFromCenter(bb3), FadeIn(lbl3), run_time=0.7)
        win.play(Wait(0.5))   #停留看结果
        win.close()   #关闭 RTM 窗口
```

这段代码在做什么：在左右两个蓝点之间画花括号——先在下方（`direction=DOWN`）并用 `get_text("distance")` 标上文字；再在同一对点的上方（`direction=UP`）画一个并改用 `get_tex(r"\Delta x")`；最后在两个竖直排列的点左侧用 `direction=LEFT` 再画一个并标 `dy`。

### `ArcBrace`

```python
def __init__(self, arc: Arc | None = None, direction: Vector3DLike = RIGHT, **kwargs) -> None:
```

| 参数 | 类型 | 默认值 | 说明 |
| --- | --- | --- | --- |
| `arc` | `Arc \| None` | `None` | 被标注的弧；`None` 时用**单位圆**（样例最后一段演示） |
| `direction` | `array-like` | `RIGHT` | 花括号朝向（弧的外侧/内侧）；样例 `RIGHT` / `LEFT` |

沿圆弧画的弧形花括号，用来标注角度/圆弧段；同样可用 `get_text` 加标签。

#### 样例：`RIGHT`/`LEFT` 两个朝向 + 自定义弧 + `arc=None`

```python
import os
import subprocess
import sys
from real_time_manim import (
    Scene, Brace, BraceText, BraceLabel, BraceBetweenPoints, ArcBrace,
    Square, Line, Dot, Text, Arc, VGroup, MathTex, DEGREES,
    BLUE, GREEN, RED, ORANGE, YELLOW, PURPLE, TEAL, GOLD, GREY_B, WHITE,
    UP, DOWN, LEFT, RIGHT, ORIGIN,
    Create, FadeIn, FadeOut, GrowFromCenter, Write,
)
from real_time_manim.vulkan_bind import MLWindow, Wait
from real_time_manim.record import fast_record_scene

class ArcBraceDemo(Scene):
    def construct(self):
        win = MLWindow(960, 540)   #创建一个 RTM 窗口
        win.scene = self   #将名为 “self” 的 manim 窗口，绑定给名为 “win” 的 RTM 窗口
        title = Text("16-ArcBrace", font_size=30, color=YELLOW).to_edge(UP)   #页面标题
        win.play(Write(title), run_time=0.5)   #写出标题

        # —— 给一段圆弧加花括号：direction=RIGHT ——
        arc = Arc(radius=1.4, start_angle=0, angle=90 * DEGREES, color=BLUE).move_to([-3.2, 0.4, 0])
        ab = ArcBrace(arc, direction=RIGHT, color=ORANGE)   #参数 “arc / direction”
        win.play(Create(arc), GrowFromCenter(ab), run_time=0.7)

        # —— direction=LEFT：甩到弧的另一侧 ——
        ab2 = ArcBrace(arc, direction=LEFT, color=TEAL)   #参数换成 LEFT
        win.play(GrowFromCenter(ab2), run_time=0.5)

        # —— 配自己的弧：更大的弧 + 标签 ——
        arc2 = Arc(radius=1.4, start_angle=90 * DEGREES, angle=90 * DEGREES, color=GREEN).move_to([3.2, 0.4, 0])
        ab3 = ArcBrace(arc2, direction=RIGHT, color=PURPLE)
        lbl = ab3.get_text("90 deg")   #方法 “get_text”
        lbl.scale(0.6).set_color(GREY_B)
        win.play(Create(arc2), GrowFromCenter(ab3), FadeIn(lbl), run_time=0.7)

        # —— arc=None：默认用单位圆 ——
        ab4 = ArcBrace(direction=RIGHT, color=RED).move_to([0, -2.2, 0])   #参数 “arc=None”
        cap = Text("arc=None（单位圆）", font_size=20, color=GREY_B).next_to(ab4, DOWN, buff=0.35)
        win.play(GrowFromCenter(ab4), FadeIn(cap), run_time=0.7)
        win.play(Wait(0.5))   #停留看结果
        win.close()   #关闭 RTM 窗口
```

这段代码在做什么：给左侧一段 90° 圆弧加花括号（`direction=RIGHT`），再在同一个弧上加一个 `direction=LEFT` 的做对比；右侧另画一段 90° 圆弧配花括号并用 `get_text("90 deg")` 标角度；最后什么都不传给 `arc`，得到一个贴着单位圆的花括号并标注 `arc=None（单位圆）`。

<video src="assets/RTM output/16-花括号标注-arcbrace.mp4" controls width="560"></video>

> 提醒：`ArcBrace` 的花括号半径跟着弧走，所以弧越短花括号越"扁"；标角度时通常把 `direction` 放在弧的外侧（默认 `RIGHT` 就是靠外）。
